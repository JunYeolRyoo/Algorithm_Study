def solution(orders, course):
    answer = []
    orders = [sorted(order) for order in orders]

    def rec(order, size, start, current, count):
        if len(current) == size:
            menu = "".join(current)
            count[menu] = count.get(menu, 0) + 1
            return

        for i in range(start, len(order)):
            current.append(order[i])
            rec(order, size, i + 1, current, count)
            current.pop()

    for size in course:
        count = {}

        for order in orders:
            rec(order, size, 0, [], count)

        most = max(count.values(), default=0)

        if most >= 2:
            for menu, cnt in count.items():
                if cnt == most:
                    answer.append(menu)

    return sorted(answer)
